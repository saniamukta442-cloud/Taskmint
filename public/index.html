/* ========================================================= TASK SYSTEM ========================================================= */ async function loadTasks(){ const list = document.getElementById("tasksList"); const loading = document.getElementById("tasksLoading"); if(!list) return; list.innerHTML = ""; if(loading){ loading.style.display = "block"; loading.textContent = "Loading tasks..."; } try{ const response = await fetch("/api/tasks",{ headers:{ "Authorization":`Bearer ${token}` } }); const data = await response.json(); if(!response.ok || !data.ok){ if(response.status === 401){ logout(); return; } throw new Error( data.error || "Could not load tasks." ); } if(loading){ loading.style.display = "none"; } if(!data.tasks || !data.tasks.length){ list.innerHTML = `
🎯
No tasks available right now.
Please check again later.
`; return; } list.innerHTML = data.tasks .map(task => renderTask(task)) .join(""); await loadMyTasks(); }catch(error){ console.error("Load tasks error:",error); if(loading){ loading.style.display = "none"; } list.innerHTML = `
⚠️ Could not load tasks.
`; } } function renderTask(task){ const reward = Number(task.reward || 0).toFixed(2); const icon = escapeHtml(task.icon || "🎯"); const title = escapeHtml(task.title || "Task"); const description = escapeHtml( task.description || "Complete this task to earn a reward." ); const type = escapeHtml( String(task.task_type || "task") .replace("_"," ") .toUpperCase() ); return `
${icon}
+$${reward}
${title}
${description}
${type}
`; } /* ========================================================= START TASK ========================================================= */ async function startTask(taskId){ const button = document.getElementById( `task-btn-${taskId}` ); const status = document.getElementById( `task-status-${taskId}` ); if(button){ button.disabled = true; button.textContent = "Starting..."; } try{ const response = await fetch( `/api/tasks/${taskId}/start`, { method:"POST", headers:{ "Content-Type":"application/json", "Authorization":`Bearer ${token}` } } ); const data = await response.json(); if(!response.ok || !data.ok){ throw new Error( data.error || "Could not start task." ); } if(button){ button.disabled = true; button.textContent = "Pending"; } if(status){ status.innerHTML = `
⏳ Task started — waiting for verification
`; } showNotice( "Task started successfully.", "success" ); await loadMyTasks(); }catch(error){ console.error( "Start task error:", error ); if(button){ button.disabled = false; button.textContent = "Start"; } if(status){ status.innerHTML = `
${escapeHtml(error.message)}
`; } } } /* ========================================================= MY TASK STATUS ========================================================= */ async function loadMyTasks(){ try{ const response = await fetch( "/api/my-tasks", { headers:{ "Authorization":`Bearer ${token}` } } ); const data = await response.json(); if(!response.ok || !data.ok){ return; } if(!Array.isArray(data.items)){ return; } data.items.forEach(item => { const button = document.getElementById( `task-btn-${item.task_id}` ); const status = document.getElementById( `task-status-${item.task_id}` ); if(!button) return; if(item.status === "completed"){ button.disabled = true; button.textContent = "Completed"; if(status){ status.innerHTML = `
✓ Completed — reward credited
`; } }else if(item.status === "pending"){ button.disabled = true; button.textContent = "Pending"; if(status){ status.innerHTML = `
⏳ Waiting for verification
`; } }else if(item.status === "rejected"){ button.disabled = false; button.textContent = "Start Again"; if(status){ status.innerHTML = `
Task was rejected
`; } } }); }catch(error){ console.error( "My tasks error:", error ); } } /* ========================================================= REFERRAL CODE ========================================================= */ function copyReferralCode(){ const element = document.getElementById( "dashReferral" ); if(!element) return; const code = element.textContent.trim(); if(!code || code === "—"){ return; } navigator.clipboard .writeText(code) .then(() => { showNotice( "Referral code copied.", "success" ); }) .catch(() => { showNotice( "Could not copy referral code.", "error" ); }); }
